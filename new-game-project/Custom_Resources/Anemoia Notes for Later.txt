when making attacks for enemies, you can create a queue of attacks by binding the Callable functions themselves
Observe:
var action_queue = []


func _ready() -> void:
	# add actions to queue
	action_queue.push_back(
		attack.bind("Lilly", 5)
	)
	action_queue.push_back(
		attack.bind("Peter", 3)
	)

	# execute actions in the order they were added
	for action in action_queue:
		action.call()

	# empty the queue
	action_queue.clear()


func attack(name: String, damage: int) -> void:
	print(name, " hit and caused ", damage, " points of damage")
